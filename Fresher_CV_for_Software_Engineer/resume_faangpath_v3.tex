\begin{rSection}{Projects}

\projecttitle{SellWise --- AI Sales Analytics \& Inventory SaaS}{\sellwiseRepo}
\vspace{2pt}
\begin{itemize}
    \item Architected a multi-tenant retail SaaS with store-isolated data access, RBAC (Owner, Manager, Salesman), and ACID POS transactions across 4 sales channels using ASP.NET Core MVC, EF Core, and SQL Server 2022 (Docker).
    \item Designed RFM customer segmentation, expense tracking, and printable invoice generation to streamline retail analytics.

    \item {\bf Stack:} C\#, ASP.NET Core MVC, EF Core, SQL Server, Docker, Python, FastAPI, Meta Prophet, Bootstrap, Chart.js, QuestPDF.
\end{itemize}
